#!/bin/bash
# 4B EXL3 decode ladder on the 2080 Ti: serve with EXL3_MR=$1, MTP k=3, fp16; kit/ladder.py c=1/2/4/8; stop
MR=${1:-2}; L=/workspace/logs/exl3-sm75; D=$L/ladder-mr$MR; rm -rf $D; mkdir -p $D; cd $D
V=/workspace/kit-work/venv-cu13
export PATH=$V/bin:$PATH CUDA_VISIBLE_DEVICES=1 EXL3_MR=$MR PYTHONPATH=/workspace/exl3-sm75b/plugin-exl3 HF_HOME=/models/hf
export VLLM_PLUGINS=lora_filesystem_resolver,lora_hf_hub_resolver,gguf,exl3 VLLM_USE_FLASHINFER_SAMPLER=0 VLLM_CACHE_ROOT=$L/vllmcache-ti-mr$MR
export EXLLAMAV3_TUNE_CACHE=$L/tune-cache.json
M=/models/hf/local/Qwen3.5-4B-exl3-6hb-3.0bpw
$V/bin/vllm serve $M --served-model-name kit --host 127.0.0.1 --port 18212 --gpu-memory-utilization 0.80 --max-model-len 16384 \
  --max-num-seqs 8 --max-num-batched-tokens 2048 --no-async-scheduling --mamba-ssm-cache-dtype float16 --mamba-cache-mode align \
  --enable-prefix-caching --limit-mm-per-prompt '{"image":0,"video":0}' --dtype float16 --attention-backend TRITON_ATTN \
  --compilation-config '{"max_cudagraph_capture_size":32}' --speculative-config '{"method":"mtp","num_speculative_tokens":3}' > server.log 2>&1 &
P=$!; t=0
until curl -sf -o /dev/null http://127.0.0.1:18212/health; do
  kill -0 $P 2>/dev/null || { echo "SERVER DIED"; tail -30 server.log; exit 1; }
  sleep 5; t=$((t+5)); [ $t -ge 900 ] && { echo TIMEOUT; kill $P; exit 1; }
done
echo "up after ${t}s"; nvidia-smi --query-gpu=index,memory.used --format=csv,noheader
grep -E "Model loading took|GPU KV cache size|EXL3|exl3_gemm" server.log | cut -c1-220 | head
$V/bin/python /workspace/kit/kit/ladder.py --url http://127.0.0.1:18212 --model kit --out $D/ladder.json --label "2080Ti-4B-exl3-mr$MR"; rc=$?
kill $P; sleep 3; pkill -P $P 2>/dev/null; for i in 1 2 3 4 5 6; do kill -0 $P 2>/dev/null || break; sleep 5; done; kill -9 $P 2>/dev/null
sleep 5; nvidia-smi --query-compute-apps=pid,used_memory --format=csv
echo "=== ladder rc=$rc"
